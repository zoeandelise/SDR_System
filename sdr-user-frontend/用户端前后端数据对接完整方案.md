```typescript
const transformDietRecordForBackend = (data: Partial<DietRecord>) => {
  const mealTypeMap = {
    'breakfast': '0',
    'lunch': '1',
    'dinner': '2', 
    'snack': '3'
  };
  
  return {
    ...data,
    mealType: mealTypeMap[data.mealType] || data.mealType,
    recordDate: data.recordTime?.split('T')[0] || new Date().toISOString().split('T')[0]
  };
};

// 后端 -> 前端
const transformDietRecordFromBackend = (data: any): DietRecord => {
  const mealTypeMap = {
    '0': 'breakfast',
    '1': 'lunch',
    '2': 'dinner',
    '3': 'snack'
  };
  
  return {
    ...data,
    mealType: mealTypeMap[data.mealType] || data.mealType,
    recordTime: data.recordDate ? `${data.recordDate}T${new Date().toTimeString().split(' ')[0]}` : new Date().toISOString()
  };
};
```

```typescript
// src/services/api.ts
const api = axios.create({
  baseURL: env.apiBaseURL,
  timeout: env.apiTimeout,
  withCredentials: true
});

// 请求拦截器 - 添加认证和请求追踪
api.interceptors.request.use(config => {
  const token = localStorage.getItem('token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  config.headers['X-Request-ID'] = `req_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`;
  return config;
});

// 响应拦截器 - 统一错误处理和数据格式化
api.interceptors.response.use(
  response => {
    // 若依框架响应格式处理
    if (response.data && typeof response.data === 'object') {
      if (response.data.code === 200 || response.data.code === undefined) {
        return response.data;
      } else {
        const error = new Error(response.data.msg || '请求失败');
        error.code = response.data.code;
        throw error;
      }
    }
    return response.data;
  },
  error => {
    // 详细错误处理
    if (error.response?.status === 401) {
      localStorage.removeItem('token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);
```

#### 3. 专业API模块

**饮食记录API：**
```typescript
export const dietRecordApi = {
  getRecords: (params: any = {}) => {
    const transformedParams = { ...params };
    if (params.date) {
      transformedParams.startDate = params.date;
      transformedParams.endDate = params.date;
      delete transformedParams.date;
    }
    return api.get(API_ENDPOINTS.diet.records, { params: transformedParams });
  },
  
  addRecord: (data: Partial<DietRecord>) => {
    const transformedData = transformDietRecordForBackend(data);
    return api.post(API_ENDPOINTS.diet.records, transformedData);
  },
  
  updateRecord: (recordId: number, data: Partial<DietRecord>) => {
    const transformedData = transformDietRecordForBackend(data);
    return api.put(`${API_ENDPOINTS.diet.records}/${recordId}`, transformedData);
  },
  
  deleteRecord: (recordId: number) => 
    api.delete(`${API_ENDPOINTS.diet.records}/${recordId}`)
};
```

## 🔄 智能同步机制

### 数据同步管理器

```typescript
// src/services/dataSync.ts
class DataSyncManager {
  private syncQueue: SyncQueueItem[] = [];
  private isOnline: boolean = navigator.onLine;
  
  // 添加操作到同步队列
  addToSyncQueue(type: 'CREATE' | 'UPDATE' | 'DELETE', endpoint: string, data?: any) {
    const item = {
      id: `sync_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
      type,
      endpoint,
      data,
      timestamp: Date.now(),
      retryCount: 0
    };
    
    this.syncQueue.push(item);
    this.saveSyncQueue();
    
    if (this.isOnline) {
      this.processSyncQueue();
    }
  }
  
  // 处理同步队列
  private async processSyncQueue() {
    if (!this.isOnline || this.syncQueue.length === 0) return;
    
    for (const item of this.syncQueue) {
      try {
        await this.syncItem(item);
        console.log(`同步成功: ${item.type} ${item.endpoint}`);
      } catch (error) {
        item.retryCount++;
        if (item.retryCount >= this.maxRetries) {
          console.error(`同步失败，丢弃: ${item.id}`);
        }
      }
    }
  }
}
```

### 智能API调用器

```typescript
class SmartApiCaller {
  // 智能创建操作 - 自动处理在线/离线状态
  async smartCreate(endpoint: string, data: any, offlineKey?: string) {
    try {
      // 尝试在线操作
      const response = await api.post(endpoint, data);
      
      // 成功后清除离线数据
      if (offlineKey) {
        this.offlineManager.removeOfflineData(offlineKey);
      }
      
      return response;
    } catch (error) {
      console.warn('在线创建失败，转为离线模式:', error);
      
      // 添加到同步队列
      const syncId = this.syncManager.addToSyncQueue('CREATE', endpoint, data);
      
      // 保存到离线存储
      if (offlineKey) {
        this.offlineManager.saveOfflineData(offlineKey, { ...data, _syncId: syncId });
      }
      
      // 返回模拟响应
      return {
        code: 200,
        msg: '已保存到本地，将在网络恢复后同步',
        data: { ...data, id: Date.now(), _offline: true, _syncId: syncId }
      };
    }
  }
}
```

## ⚠️ 错误处理策略

### 分层错误处理

#### 1. 网络层错误处理
```typescript
// 网络连接错误
if (error.code === 'ECONNABORTED' || error.message.includes('timeout')) {
  return '网络连接超时，请检查网络连接后重试';
}

// 服务器错误
switch (error.response?.status) {
  case 401: return '认证失败，请重新登录';
  case 403: return '权限不足，无法访问';
  case 404: return '请求的资源不存在';
  case 500: return '服务器内部错误，请稍后重试';
}
```

#### 2. 业务层错误处理
```typescript
// 若依框架业务错误
if (response.data.code !== 200) {
  const error = new Error(response.data.msg || '请求失败');
  error.code = response.data.code;
  throw error;
}
```
